🔍 How Hybrid Retrieval Works

class HybridRetrievalEngine {
  retrieve(query, k = 5) {
    // 1. Expand query with synonyms
    const expanded = this.expandQuery(query);

    // 2. Score every chunk with TF-IDF
    const tfidfScores = chunks.map(c => computeTfidf(expanded, c));

    // 3. Score with 128-dim cosine similarity
    const qEmbed = this.embedder.embed(expanded);
    const semScores = chunks.map(c => 
      cosineSimilarity(qEmbed, c.embedding)
    );

    // 4. Weighted fusion & re-rank
    const hybrid = chunks.map((c, i) => ({
      ...c,
      score: (1 - semanticWeight) * tfidfScores[i]
           + semanticWeight * semScores[i]
    }));

    return hybrid.sort((a, b) => b.score - a.score).slice(0, k);
  }
}

Quick Start

Three Lines to Intelligence

Drop the script. Load your data. The agent is live. No build step. No server changes.

index.html html 
<!-- 1. Include RAGina T2 -->
<script src="https://cdn.jsdelivr.net/gh/suryasticsai/RAGina@main/ragina-t2.js"></script>

<script>
  // 2. Load your business knowledge
  RAGina.loadData({
    "pricing.pdf": { bodyText: "Enterprise plan: $99/mo..." },
    "faq.md":    { bodyText: "Refunds processed in 5-7 days..." },
    "policy.html":{ bodyText: "SLA: 99.9% uptime guarantee..." }
  });

  // 3. Register a custom business tool
  RAGina.registerTool('lookupOrder', {
    description: 'Look up an order by ID from internal API',
    parameters: { orderId: 'string, e.g. ORD-2026-8842' },
    handler: async ({ orderId }) => {
      const r = await fetch(`/api/orders/${orderId}`);
      return r.ok ? await r.json() : 'Order not found';
    }
  });

  // That's it — the chat bubble is live with full T2 capabilities
</script>
headless-mode.js javascript 
// No UI — use RAGina's brain inside your own dashboard
const { answer, steps } = await RAGina.query(
  'What is the refund policy for enterprise customers?',
  { contextText: 'Enterprise SLA docs...', topK: 5 }
);

console.log(answer);   // "Enterprise refunds are processed in 5-7 business days..."
console.log(steps);    // Full trace: which chunks retrieved, which tools called
